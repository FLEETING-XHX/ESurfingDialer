cmake_minimum_required(VERSION 3.20)
project(ESurfingDialerNative VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # glibc armhf otherwise keeps a 32-bit time_t even on current Debian.
  # Both macros are required for its time64 ABI; musl already uses 64-bit time.
  add_compile_definitions(_FILE_OFFSET_BITS=64 _TIME_BITS=64)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_program(JQ_EXECUTABLE jq)
  if(JQ_EXECUTABLE)
    add_test(NAME health-regression COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/health-regression.sh ${CMAKE_CURRENT_SOURCE_DIR}/docker/healthcheck.sh)
    add_test(NAME watchdog-regression COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/watchdog-regression.sh)
    set_tests_properties(health-regression watchdog-regression PROPERTIES TIMEOUT 180)
  else()
    message(STATUS "jq unavailable: shell supervision tests require separate execution")
  endif()
endif()
option(NATIVE_SANITIZERS "Enable AddressSanitizer/UBSan in isolated tests" OFF)
if(NATIVE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Docker Desktop's current ASLR/PIE layout crashes even an empty ASan main.
    # Non-PIE is scoped to sanitizer test builds; runtime Release keeps default PIE.
    add_compile_options(-fno-pie)
    add_link_options(-no-pie)
  endif()
endif()
file(GLOB CIPHER_SOURCES CONFIGURE_DEPENDS third_party/ciphers/algo/*.c)
file(GLOB EVP_SOURCES CONFIGURE_DEPENDS third_party/ciphers/sim/*.c)
add_library(reference_ciphers STATIC ${CIPHER_SOURCES} ${EVP_SOURCES} third_party/ciphers/CipherUtils.c third_party/ciphers/KeyData.c third_party/ciphers/factory.c)
target_include_directories(reference_ciphers PUBLIC third_party/ciphers/include)
if(MSVC)
  target_compile_options(reference_ciphers PRIVATE /utf-8)
  target_compile_definitions(reference_ciphers PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
find_package(Threads REQUIRED)
add_library(dialer_core src/core.cpp src/state_store.cpp src/cipher.cpp src/config.cpp src/state_json.cpp src/identity.cpp src/planner.cpp src/portal_policy.cpp src/log.cpp)
target_link_libraries(dialer_core PUBLIC reference_ciphers Threads::Threads)
target_include_directories(dialer_core PUBLIC include)
if(MSVC)
  target_compile_options(dialer_core PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(dialer_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(esurfing-native src/main.cpp)
target_link_libraries(esurfing-native PRIVATE dialer_core)
enable_testing()
add_test(NAME cli-regression COMMAND ${CMAKE_COMMAND} -DBINARY=$<TARGET_FILE:esurfing-native> -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-fixtures -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_regression.cmake)
set_tests_properties(cli-regression PROPERTIES TIMEOUT 15)
add_executable(native-regression tests/regression.cpp)
target_link_libraries(native-regression PRIVATE dialer_core)
add_test(NAME native-regression COMMAND native-regression)
add_executable(identity-regression tests/identity_regression.cpp)
target_link_libraries(identity-regression PRIVATE dialer_core)
add_test(NAME identity-regression COMMAND identity-regression)
add_executable(planner-regression tests/planner_regression.cpp)
target_link_libraries(planner-regression PRIVATE dialer_core)
add_test(NAME planner-regression COMMAND planner-regression)
add_executable(log-regression tests/log_regression.cpp)
target_link_libraries(log-regression PRIVATE dialer_core)
add_test(NAME log-regression COMMAND log-regression)
set_tests_properties(log-regression PROPERTIES TIMEOUT 20)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(log-pipe-regression tests/log_pipe_regression.cpp)
  target_link_libraries(log-pipe-regression PRIVATE dialer_core)
  add_test(NAME log-pipe-regression COMMAND log-pipe-regression)
  set_tests_properties(log-pipe-regression PROPERTIES TIMEOUT 20)
endif()
add_executable(cipher-regression tests/cipher_regression.cpp)
target_link_libraries(cipher-regression PRIVATE dialer_core)
add_test(NAME cipher-regression COMMAND cipher-regression)
add_executable(cipher-negative tests/cipher_negative.cpp)
target_link_libraries(cipher-negative PRIVATE dialer_core)
add_test(NAME cipher-negative COMMAND cipher-negative)
target_compile_definitions(cipher-regression PRIVATE REFERENCE_VECTORS="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference-vectors.tsv")
set_tests_properties(native-regression identity-regression planner-regression cipher-regression cipher-negative PROPERTIES TIMEOUT 30)
if(WIN32)
  set(NETWORK_DEFAULT OFF)
else()
  set(NETWORK_DEFAULT ON)
endif()
option(NATIVE_NETWORK "Build libcurl/libxml2 protocol and HTTP implementation" ${NETWORK_DEFAULT})
if(NATIVE_NETWORK)
  find_package(Threads REQUIRED)
  find_package(CURL REQUIRED)
  find_package(LibXml2 REQUIRED)
  add_library(dialer_network src/protocol.cpp src/http.cpp src/session.cpp src/network_monitor.cpp)
  target_link_libraries(dialer_network PUBLIC dialer_core CURL::libcurl LibXml2::LibXml2 Threads::Threads)
  add_executable(protocol-regression tests/protocol_regression.cpp)
  target_link_libraries(protocol-regression PRIVATE dialer_network)
  add_test(NAME protocol-regression COMMAND protocol-regression)
  set_tests_properties(protocol-regression PROPERTIES TIMEOUT 30)
  add_executable(portal-compatibility tests/portal_compatibility.cpp)
  target_link_libraries(portal-compatibility PRIVATE dialer_network)
  add_test(NAME portal-compatibility COMMAND portal-compatibility)
  set_tests_properties(portal-compatibility PROPERTIES TIMEOUT 30)
  add_executable(network-regression tests/network_regression.cpp)
  target_link_libraries(network-regression PRIVATE dialer_network)
  add_test(NAME network-regression COMMAND network-regression)
  set_tests_properties(network-regression PROPERTIES TIMEOUT 10)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # Separate fixture executable; never copied into the runtime image. The actual
    # production executable retains its loopback restriction regardless of env.
    add_executable(native-runtime-fixture src/main.cpp src/runtime.cpp)
    target_compile_definitions(native-runtime-fixture PRIVATE ESURFING_NETWORK=1 ESURFING_RUNTIME_FIXTURE=1)
    target_link_libraries(native-runtime-fixture PRIVATE dialer_network)
    add_executable(runtime-integration tests/runtime_integration.cpp)
    target_link_libraries(runtime-integration PRIVATE dialer_network)
    add_test(NAME runtime-integration COMMAND runtime-integration $<TARGET_FILE:native-runtime-fixture>)
    set_tests_properties(runtime-integration PROPERTIES TIMEOUT 40)
    add_test(NAME runtime-paused-network COMMAND runtime-integration $<TARGET_FILE:native-runtime-fixture> --paused)
    set_tests_properties(runtime-paused-network PROPERTIES TIMEOUT 40)
    add_test(NAME runtime-log-failure COMMAND runtime-integration $<TARGET_FILE:native-runtime-fixture> --log-failure)
    set_tests_properties(runtime-log-failure PROPERTIES TIMEOUT 40)
  endif()
  if(UNIX)
    add_executable(http-regression tests/http_regression.cpp)
    target_link_libraries(http-regression PRIVATE dialer_network Threads::Threads)
    add_test(NAME http-regression COMMAND http-regression)
    set_tests_properties(http-regression PROPERTIES TIMEOUT 30)
  endif()
  target_sources(esurfing-native PRIVATE src/runtime.cpp)
  target_compile_definitions(esurfing-native PRIVATE ESURFING_NETWORK=1)
  target_link_libraries(esurfing-native PRIVATE dialer_network Threads::Threads)
endif()
