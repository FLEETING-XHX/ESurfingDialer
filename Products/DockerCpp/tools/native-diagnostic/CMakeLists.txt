cmake_minimum_required(VERSION 3.20)
project(RecoveredModuleCipherDiagnostics LANGUAGES CXX)
add_library(recovered-module-cipher STATIC recovered_cipher.cpp ../../src/native_cipher.cpp)
target_compile_features(recovered-module-cipher PUBLIC cxx_std_20)
target_include_directories(recovered-module-cipher PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
add_executable(recovered-cipher-check recovered_cipher_cli.cpp)
target_link_libraries(recovered-cipher-check PRIVATE recovered-module-cipher)
foreach(target recovered-module-cipher recovered-cipher-check)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
