# Bind each build to the exact source inputs and resulting executable.
set(NATIVE_INPUT_FILES CMakeLists.txt Dockerfile)
foreach(directory src include third_party profiles docker cmake scripts)
  file(GLOB_RECURSE inputs CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${directory}/*")
  foreach(input IN LISTS inputs)
    if(NOT input MATCHES "/__pycache__/|[.]pyc$" AND NOT IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${input}")
      list(APPEND NATIVE_INPUT_FILES "${input}")
    endif()
  endforeach()
endforeach()
list(SORT NATIVE_INPUT_FILES)
set(input_json "")
set(input_separator "")
foreach(input IN LISTS NATIVE_INPUT_FILES)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${input}" digest)
  string(APPEND input_json "${input_separator}\"${input}\":\"${digest}\"")
  set(input_separator ",")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${input}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source-inputs.json"
  "{\"schemaVersion\":1,\"version\":\"${PROJECT_VERSION}\",\"channel\":\"${NATIVE_RELEASE_CHANNEL}\",\"inputs\":{${input_json}}}\n")
add_custom_target(native-build-manifest ALL
  COMMAND ${CMAKE_COMMAND} "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/source-inputs.json"
    "-DBINARY=$<TARGET_FILE:esurfing-native>" "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/build-source.json"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/write-build-manifest.cmake"
  DEPENDS esurfing-native VERBATIM)
