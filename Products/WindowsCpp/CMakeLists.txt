cmake_minimum_required(VERSION 3.24)
project(ESurfingDialerWindowsCpp VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "WindowsCpp UI requires Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(client_services src/ClientServices.cpp)
target_include_directories(client_services PUBLIC include)
find_package(Qt6 6.8 REQUIRED COMPONENTS Quick QuickControls2 Svg Network Test)
qt_standard_project_setup(REQUIRES 6.8)
add_library(local_client_services src/LocalClientServices.cpp)
set(docker_ciphers "${CMAKE_CURRENT_SOURCE_DIR}/../DockerCpp/third_party/ciphers")
set(docker_port "${CMAKE_CURRENT_SOURCE_DIR}/vendor/dockercpp")
add_library(auth_session_2104
  src/AuthSession2104.cpp src/AuthHttpTransport.cpp src/CampusPortal.cpp src/AuthWorker.cpp src/Native2082Cipher.cpp
  "${docker_port}/src/module_loader.cpp" "${docker_port}/src/native_cipher.cpp"
  "${docker_port}/src/sm4_cipher.cpp" "${docker_port}/src/aes_cipher.cpp"
  "${docker_port}/third_party/ciphers/algo/zuc.c" "${docker_ciphers}/CipherUtils.c"
  "${docker_ciphers}/algo/sm4_common.c" "${docker_ciphers}/sim/SimEvpAes.c" vendor/lzma/LzmaDec.c)
target_include_directories(auth_session_2104 PUBLIC include)
target_include_directories(auth_session_2104 PRIVATE "${docker_port}/include" "${docker_ciphers}/include"
  "${docker_ciphers}/algo" vendor/lzma)
target_link_libraries(auth_session_2104 PUBLIC Qt6::Core Qt6::Network)
set(loader_data "${CMAKE_CURRENT_SOURCE_DIR}/../Docker/src/main/resources/libdaproxy.so")
qt_add_resources(auth_session_2104 android2082_profile PREFIX "/auth2082"
  BASE "${CMAKE_CURRENT_SOURCE_DIR}/../Docker/src/main/resources" FILES "${loader_data}")
set_source_files_properties("${docker_ciphers}/CipherUtils.c" PROPERTIES COMPILE_DEFINITIONS "_CRT_SECURE_NO_WARNINGS")
target_link_libraries(local_client_services PUBLIC client_services auth_session_2104 PRIVATE crypt32)
if(MSVC)
  target_compile_options(local_client_services PRIVATE /utf-8 /W4 /permissive- /EHsc)
  target_compile_options(auth_session_2104 PRIVATE /utf-8 /W4 $<$<COMPILE_LANGUAGE:CXX>:/permissive-;/EHsc>)
endif()
qt_add_library(client_bridge STATIC src/ClientBridge.cpp include/ClientBridge.h)
target_link_libraries(client_bridge PUBLIC client_services Qt6::Core)
target_include_directories(client_bridge PUBLIC include)
qt_add_executable(ESurfingDialerWindowsCpp WIN32 src/QuickMain.cpp)
target_link_libraries(ESurfingDialerWindowsCpp PRIVATE client_bridge local_client_services Qt6::Quick Qt6::QuickControls2 Qt6::Svg dwmapi)
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(ESurfingDialerWindowsCpp URI ESurfing VERSION 1.0 QML_FILES
  qml/Main.qml qml/Theme.qml qml/UiText.qml qml/Icon.qml qml/Card.qml qml/PageScroll.qml
  qml/ActionButton.qml qml/ToggleRow.qml qml/HomePage.qml qml/SettingsPage.qml
  qml/AccountSheet.qml qml/LogPage.qml qml/FormField.qml)
file(GLOB icon_assets CONFIGURE_DEPENDS "assets/icons/*.svg")
qt_add_resources(ESurfingDialerWindowsCpp icons PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ${icon_assets})
set(font_dir "${CMAKE_CURRENT_SOURCE_DIR}/../Windows/app/ESurfingDialerLite/Fonts")
option(WINDOWSCPP_COMPRESS_FONTS "Trade installer size for smaller embedded fonts on disk" OFF)
set(font_resource_options)
if(WINDOWSCPP_COMPRESS_FONTS)
  list(APPEND font_resource_options -compress-algo zlib -compress 9 -threshold 20)
endif()
qt_add_resources(ESurfingDialerWindowsCpp fonts PREFIX "/fonts" BASE "${font_dir}"
  FILES "${font_dir}/NotoSansSC-Regular.ttf" "${font_dir}/NotoSansSC-Bold.ttf"
  OPTIONS ${font_resource_options})
if(MSVC)
  target_compile_options(client_bridge PRIVATE /utf-8 /W4 /permissive- /EHsc)
endif()
if(MSVC)
  target_compile_options(client_services PRIVATE /utf-8 /W4 /permissive- /EHsc)
  target_compile_options(ESurfingDialerWindowsCpp PRIVATE /utf-8 /W4 /permissive- /EHsc)
  target_link_options(ESurfingDialerWindowsCpp PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(auth_bootstrap_probe src/BootstrapProbe.cpp)
  target_link_libraries(auth_bootstrap_probe PRIVATE auth_session_2104)
  add_test(NAME auth_bootstrap_probe_help COMMAND auth_bootstrap_probe --help)
  add_executable(service_contract_tests tests/ServiceTests.cpp)
  target_link_libraries(service_contract_tests PRIVATE client_services)
  if(MSVC)
    target_compile_options(service_contract_tests PRIVATE /utf-8 /W4 /permissive- /EHsc)
  endif()
  add_test(NAME service_contract COMMAND service_contract_tests)
  qt_add_executable(bridge_tests tests/BridgeTests.cpp)
  target_link_libraries(bridge_tests PRIVATE client_bridge Qt6::Test)
  if(MSVC)
    target_compile_options(bridge_tests PRIVATE /utf-8 /W4 /permissive- /EHsc)
  endif()
  add_test(NAME bridge_contract COMMAND bridge_tests)
  add_executable(local_service_tests tests/LocalServiceTests.cpp)
  target_link_libraries(local_service_tests PRIVATE local_client_services Qt6::Test crypt32)
  if(MSVC)
    target_compile_options(local_service_tests PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_test(NAME local_service_contract COMMAND local_service_tests)
  add_executable(auth_session_tests tests/AuthSessionTests.cpp)
  target_link_libraries(auth_session_tests PRIVATE auth_session_2104 Qt6::Test)
  if(MSVC)
    target_compile_options(auth_session_tests PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_test(NAME auth_session_offline COMMAND auth_session_tests)
  add_test(NAME qml_smoke COMMAND ESurfingDialerWindowsCpp --smoke-test)
  set_tests_properties(qml_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
endif()
install(TARGETS ESurfingDialerWindowsCpp RUNTIME DESTINATION .)
install(FILES assets/icons/LICENSE.txt DESTINATION licenses RENAME Lucide.txt)
install(FILES "${font_dir}/NotoSansSC-OFL.txt" DESTINATION licenses)
